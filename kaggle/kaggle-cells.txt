# bolt.diy — Kaggle GPU Model Server Setup

**Qwen3.6-35B-A3B (GGUF)** ko Kaggle T4 GPU par llama.cpp se chalata hai aur
Cloudflare Tunnel + Worker ke through bolt.diy se connect karta hai.

**Flow:** `bolt.diy` → `Cloudflare Worker (permanent URL)` → `Kaggle Tunnel (dynamic URL)` → `Model`

| Cell | Kaam | Kab chalana |
|------|------|-------------|
| 1 | System check | Fresh session (optional) |
| 2 | Dependencies install | Fresh session |
| 3 | Model download (~21 GB) | Sirf pehli baar |
| 4 | Model file verify | Download ke baad |
| 5 | Model load (2-3 min) | Har restart |
| 6 | Server script create | Fresh session |
| 7 | Server + Tunnel + Worker auto-update | Har restart (band mat karna) |

**Restart routine:** Cell 2 → Cell 5 → Cell 7. Model cache mein rehta hai (`/kaggle/tmp/models/`).

**Cell 7 mein apni values daalna mat bhulna:**
- `CF_ACCOUNT_ID` — dash.cloudflare.com → Workers & Pages → Overview → right sidebar
- `CF_API_TOKEN` — My Profile → API Tokens → **Account / Workers Scripts / Edit** permission

from huggingface_hub import hf_hub_download
import os

SAVE_PATH  = "/kaggle/tmp/models"
FILE_NAME  = "Qwen3.6-35B-A3B-Uncensored-HauhauCS-Aggressive-Q4_K_M.gguf"
FINAL_PATH = os.path.join(SAVE_PATH, FILE_NAME)
REPO_ID    = "HauhauCS/Qwen3.6-35B-A3B-Uncensored-HauhauCS-Aggressive"

os.makedirs(SAVE_PATH, exist_ok=True)

if os.path.exists(FINAL_PATH):
    size_gb = os.path.getsize(FINAL_PATH) / (1024**3)
    if 20.0 <= size_gb <= 22.5:
        print(f"Model already downloaded! Size: {size_gb:.2f} GB")
        MODEL_PATH = FINAL_PATH
        print(f"MODEL_PATH = {MODEL_PATH}")
        # Skip download
        import sys
        print("Skipping download — use Cell 4 to verify")
    else:
        print(f"Incomplete file ({size_gb:.2f} GB) — deleting...")
        os.remove(FINAL_PATH)
else:
    print(f"Starting fresh download...")
    print(f"File: {FILE_NAME}")
    print(f"Saving to: {SAVE_PATH}")
    print(f"ETA: 15-40 min\n")

    hf_hub_download(
        repo_id=REPO_ID,
        filename=FILE_NAME,
        local_dir=SAVE_PATH,
        local_dir_use_symlinks=False,
        force_download=False,
        resume_download=True,
    )

if os.path.exists(FINAL_PATH):
    size_gb = os.path.getsize(FINAL_PATH) / (1024**3)
    print(f"\nDownload complete! Size: {size_gb:.2f} GB")
    if 20.0 <= size_gb <= 22.5:
        print("File complete — ready to load!")
        MODEL_PATH = FINAL_PATH
    else:
        print(f"Size unexpected ({size_gb:.2f} GB). Retry.")
else:
    print("Download failed!")

from llama_cpp import Llama
import os
import builtins

MODEL_PATH = "/kaggle/tmp/models/Qwen3.6-35B-A3B-Uncensored-HauhauCS-Aggressive-Q4_K_M.gguf"

if not os.path.exists(MODEL_PATH):
    import subprocess
    res = subprocess.run(
        ["find", "/kaggle", "/tmp", "-name", "*Qwen3.6*Q4_K_M*.gguf", "-type", "f"],
        capture_output=True, text=True
    )
    if res.stdout.strip():
        MODEL_PATH = res.stdout.strip().split("\n")[0]
        print(f"Found: {MODEL_PATH}")
    else:
        raise FileNotFoundError("Model not found! Run Cell 3.")

print(f"Loading model: {MODEL_PATH}")
print("First load: 1-3 minutes...")

llm = Llama(
    model_path=MODEL_PATH,
    n_gpu_layers=-1,
    n_ctx=8192,
    n_batch=512,
    n_threads=8,
    verbose=False,
    seed=42,
    use_mmap=True,
    use_mlock=False,
)

builtins.llm = llm
globals()["llm"] = llm

print("\nModel loaded!")
print(f"Context: {llm.n_ctx()}")

# Smoke test
out = llm(
    "<|im_start|>user\nSay hello in one sentence.<|im_end|>\n<|im_start|>assistant\n",
    max_tokens=30, temperature=0.7, echo=False
)
print("Smoke test:", out["choices"][0]["text"].strip())
print("Ready for server!")

# CELL 6: FASTAPI SERVER SCRIPT
# ============================================
import os

server_code = r'''
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import StreamingResponse
from pydantic import BaseModel
from typing import List, Optional
import uvicorn, time, json, asyncio

app = FastAPI(
    title="Beast AI Server",
    description="Qwen3.6-35B Aggressive via Kaggle GPU",
    version="1.0.0"
)

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

class Message(BaseModel):
    role: str
    content: str

class ChatRequest(BaseModel):
    model: Optional[str] = "qwen3.6-35b-aggressive"
    messages: List[Message]
    max_tokens: Optional[int] = 2048
    temperature: Optional[float] = 0.7
    top_p: Optional[float] = 0.9
    stream: Optional[bool] = False
    stop: Optional[List[str]] = None

def messages_to_prompt(messages):
    prompt = ""
    for msg in messages:
        role = msg.role if msg.role in ("system", "user", "assistant") else "user"
        prompt += f"<|im_start|>{role}\n{msg.content}<|im_end|>\n"
    prompt += "<|im_start|>assistant\n"
    return prompt

@app.get("/")
@app.get("/health")
async def health():
    return {
        "status": "alive",
        "model": "Qwen3.6-35B-A3B-Uncensored-Q4_K_M",
        "platform": "Kaggle GPU T4"
    }

@app.get("/v1/models")
async def list_models():
    return {
        "object": "list",
        "data": [{
            "id": "qwen3.6-35b-aggressive",
            "object": "model",
            "created": int(time.time()),
            "owned_by": "HauhauCS",
            "permission": [],
            "root": "qwen3.6-35b-aggressive"
        }]
    }

@app.post("/v1/chat/completions")
async def chat_completions(request: ChatRequest):
    try:
        prompt = messages_to_prompt(request.messages)
        stop_tokens = request.stop or [
            "<|im_end|>", "<|im_start|>", "<|endoftext|>"
        ]

        if request.stream:
            async def stream_generator():
                response_id = f"chatcmpl-{int(time.time())}"
                first = {
                    "id": response_id, "object": "chat.completion.chunk",
                    "created": int(time.time()), "model": request.model,
                    "choices": [{"index": 0, "delta": {"role": "assistant"}, "finish_reason": None}]
                }
                yield f"data: {json.dumps(first)}\n\n"

                stream = llm(
                    prompt,
                    max_tokens=request.max_tokens or 2048,
                    temperature=request.temperature or 0.7,
                    top_p=request.top_p or 0.9,
                    stop=stop_tokens,
                    echo=False, stream=True,
                )

                for chunk in stream:
                    token = chunk["choices"][0].get("text", "")
                    if token:
                        data = {
                            "id": response_id, "object": "chat.completion.chunk",
                            "created": int(time.time()), "model": request.model,
                            "choices": [{"index": 0, "delta": {"content": token}, "finish_reason": None}]
                        }
                        yield f"data: {json.dumps(data)}\n\n"
                        await asyncio.sleep(0)

                final = {
                    "id": response_id, "object": "chat.completion.chunk",
                    "created": int(time.time()), "model": request.model,
                    "choices": [{"index": 0, "delta": {}, "finish_reason": "stop"}]
                }
                yield f"data: {json.dumps(final)}\n\n"
                yield "data: [DONE]\n\n"

            return StreamingResponse(
                stream_generator(),
                media_type="text/event-stream",
                headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"}
            )

        output = llm(
            prompt,
            max_tokens=request.max_tokens or 2048,
            temperature=request.temperature or 0.7,
            top_p=request.top_p or 0.9,
            stop=stop_tokens,
            echo=False,
        )

        response_text = output["choices"][0]["text"].strip()
        usage = output.get("usage", {})

        return {
            "id": f"chatcmpl-{int(time.time())}",
            "object": "chat.completion",
            "created": int(time.time()),
            "model": request.model,
            "choices": [{
                "index": 0,
                "message": {"role": "assistant", "content": response_text},
                "finish_reason": "stop"
            }],
            "usage": {
                "prompt_tokens": usage.get("prompt_tokens", 0),
                "completion_tokens": usage.get("completion_tokens", 0),
                "total_tokens": usage.get("total_tokens", 0)
            }
        }

    except Exception as e:
        import traceback
        traceback.print_exc()
        raise HTTPException(status_code=500, detail=str(e))

if __name__ == "__main__":
    uvicorn.run(app, host="0.0.0.0", port=8000, workers=1)
'''

with open("/kaggle/working/server.py", "w") as f:
    f.write(server_code)

print("Server script saved: /kaggle/working/server.py")

# CELL 7: SERVER + CLOUDFLARE TUNNEL + AUTO UPDATE
# IMPORTANT: Apna CF_ACCOUNT_ID aur CF_API_TOKEN daalo
# ============================================

import subprocess, threading, time, re, sys
import builtins, importlib.util, uvicorn, os, requests

# ══════════════════════════════════════════════
# APNI VALUES YAHAN DAALO
CF_ACCOUNT_ID  = "APNA_CLOUDFLARE_ACCOUNT_ID"   # Dashboard se lo
CF_WORKER_NAME = "kaggle-ai-proxy"               # Worker ka naam
CF_API_TOKEN   = "APNA_CF_API_TOKEN"             # Edit Workers permission wala
# ══════════════════════════════════════════════

if not hasattr(builtins, "llm") or builtins.llm is None:
    raise RuntimeError("llm not loaded! Run Cell 5 first.")

sys.path.insert(0, "/kaggle/working")

spec = importlib.util.spec_from_file_location("server", "/kaggle/working/server.py")
server_module = importlib.util.module_from_spec(spec)
server_module.llm = builtins.llm
sys.modules["server"] = server_module
spec.loader.exec_module(server_module)

def run_server():
    uvicorn.run(
        server_module.app,
        host="0.0.0.0",
        port=8000,
        log_level="warning",
        access_log=False
    )

server_thread = threading.Thread(target=run_server, daemon=True)
server_thread.start()
print("FastAPI server starting on port 8000...")
time.sleep(5)

try:
    r = requests.get("http://127.0.0.1:8000/health", timeout=5)
    print(f"Local health: {r.status_code} — {r.json()['status']}")
except Exception as e:
    print(f"Local check: {e}")

print("\nStarting Cloudflare tunnel...")
cf_bin = "/kaggle/working/cloudflared"

cf_process = subprocess.Popen(
    [cf_bin, "tunnel", "--url", "http://localhost:8000", "--no-autoupdate"],
    stdout=subprocess.PIPE,
    stderr=subprocess.STDOUT,
    text=True, bufsize=1
)

cloudflare_url = None
print("Waiting for URL (10-30 sec)...")

for i, line in enumerate(cf_process.stdout):
    if "trycloudflare.com" in line:
        urls = re.findall(r'https://[a-zA-Z0-9\-]+\.trycloudflare\.com', line)
        if urls:
            cloudflare_url = urls[0]
            break
    if i > 80:
        break

def update_cloudflare_worker(tunnel_url: str):
    """Cloudflare Worker ka KAGGLE_TUNNEL_URL auto update karo"""
    url = (
        f"https://api.cloudflare.com/client/v4/accounts/"
        f"{CF_ACCOUNT_ID}/workers/scripts/{CF_WORKER_NAME}/secrets"
    )
    payload = {
        "name": "KAGGLE_TUNNEL_URL",
        "text": tunnel_url,
        "type": "secret_text"
    }
    headers = {
        "Authorization": f"Bearer {CF_API_TOKEN}",
        "Content-Type": "application/json"
    }
    try:
        r = requests.put(url, json=payload, headers=headers, timeout=15)
        if r.status_code in [200, 201]:
            print(f"Worker URL updated: {tunnel_url}")
            return True
        else:
            print(f"Worker update failed: {r.status_code}")
            print(f"Manual update karo Dashboard pe!")
            return False
    except Exception as e:
        print(f"Auto-update error: {e}")
        print(f"Manual update karo: {tunnel_url}")
        return False

print()
print("=" * 70)
print("  BEAST AI SERVER — LIVE!")
print("=" * 70)

if cloudflare_url:
    print(f"\nKaggle Tunnel URL:\n    {cloudflare_url}")
    print(f"\nAuto-updating Cloudflare Worker...")
    success = update_cloudflare_worker(cloudflare_url)

    if success:
        print(f"\nBOLT.DIY SETTINGS (PERMANENT — KABHI NAHI BADLEGA):")
        print(f"    Base URL : https://{CF_WORKER_NAME}.CodeEmpireHub.workers.dev/v1")
        print(f"    API Key  : sk-no-key")
        print(f"    Model    : qwen3.6-35b-aggressive")
    else:
        print(f"\nManual Worker update:")
        print(f"    Dashboard → kaggle-ai-proxy → Settings → Variables")
        print(f"    KAGGLE_TUNNEL_URL = {cloudflare_url}")
else:
    print("URL nahi mila — CF logs check karo!")

print()
print("=" * 70)
print(f"    GET  /health")
print(f"    GET  /v1/models")
print(f"    POST /v1/chat/completions")
print("=" * 70)
print()
print("Keep-alive started (4-8 hours session)")
print("Is cell ko band mat karna!")
print()

start = time.time()
session_hours = 8
check_interval = 180
heartbeat = 0

while True:
    time.sleep(check_interval)
    heartbeat += 1
    elapsed   = (time.time() - start) / 3600
    remaining = session_hours - elapsed
    alive     = server_thread.is_alive()

    print(
        f"[{heartbeat}] "
        f"Elapsed: {elapsed:.1f}h | "
        f"Remaining: {remaining:.1f}h | "
        f"Server: {'OK' if alive else 'DEAD'}"
    )

    if remaining <= 0:
        print("Time limit reached. Restart notebook.")
        break

    if not alive:
        print("Server crash — restarting...")
        server_thread = threading.Thread(target=run_server, daemon=True)
        server_thread.start()
        time.sleep(4)
        print("Server restarted!")
